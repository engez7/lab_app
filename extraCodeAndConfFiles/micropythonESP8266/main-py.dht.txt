import network
from umqtt.simple import MQTTClient
from machine import Pin, reset
import time
import dht 

# Set client STATIC IP!!
# Example only: adapt these to your own local network (typically 192.168.x.x).
CLIENT_IP = "192.168.1.102"

#CLIENT_HOST_NAME = "DHT_z01_n02"    #NON FUNZIONA: LA VODAFONE STATION NON VEDE IL NOME DELL'ESP8266!!!

CLIENT_IP_NETMASK = "255.255.255.0"
ROUTER_IP = "192.168.1.1"
DNS_IP = "8.8.8.8"

MQTT_CLIENT_ID = "esp_client_z01_n02" #client zone 01 number 02
MQTT_SERVER = "192.168.1.254"  # example only: your MQTT broker's local IP
MQTT_TOPIC_TEMPHUM = b"esp/dht/TempHum_z01" # temperature and humidity zone 01
mqtt_connected = False # flag for... mqtt is connected or not

WIFI_SSID = "YOUR_WIFI_SSID"
WIFI_PASSWORD = "YOUR_WIFI_PASSWORD"

# To wait to publish at least 5 s...
last_message = 0
message_interval = 5

led = Pin(2, Pin.OUT) #ESP-8622 Wemos -> GPIO2 = D4+led
sensor = dht.DHT22(Pin(0)) #DHT-22 -> GPIO0 = D3

# WiFi connectivity function.
def connectWIFI():
    # Create interface AP Access Point and disable it
    ap_if = network.WLAN(network.AP_IF)
    ap_if.active(False)
    # Create interface as STAtion
    sta_if = network.WLAN(network.STA_IF)
    print('WIFI connection: ', sta_if.isconnected())
    
    if not sta_if.isconnected():
        print('Connecting to wifi...')
        # Enable the interface
        sta_if.active(True)
        # Connect
        sta_if.ifconfig((CLIENT_IP,CLIENT_IP_NETMASK,ROUTER_IP,DNS_IP))
        #sta_if.config(dhcp_hostname = CLIENT_HOST_NAME)        #NON FUNZIONA: LA VODAFONE STATION NON VEDE IL NOME DELL'ESP8266!!!
        sta_if.connect(WIFI_SSID,WIFI_PASSWORD)
        # Wait till connection is established
        while not sta_if.isconnected():
            pass

        print("Wifi connected")  
        print(sta_if.ifconfig())

def connectMQTT(c):
    global mqtt_connected
    while True:
        try:
            print("MQTT connecting...")
            c.connect()
            mqtt_connected = True
            print("MQTT connected and ready to publish message")
            # Led pulse before OFF if ESP-8266 connected to wifi and to MQTT broker
            led.value(1)
            time.sleep(0.3)
            led.value(0)
            time.sleep(0.3)
            led.value(1)
            time.sleep(0.3)
            led.value(0)
            time.sleep(0.3)
            led.value(1)
            break
        except OSError as e:
            print("MQTT connection failed, retrying...")
            mqtt_connected = False
            time.sleep(10)

def read_sensor():
    try:
        sensor.measure()
        temp = sensor.temperature()
        hum = sensor.humidity()
        if (isinstance(temp, float) and isinstance(hum, float)) or (isinstance(temp, int) and isinstance(hum, int)):
            temp = ('{0:3.1f}'.format(temp))
            hum =  ('{0:3.1f}'.format(hum))
            return temp, hum
        else:
            return('Invalid sensor readings.')
    except OSError as e:
        return('Failed to read sensor.')

try:
    # Led ON during wifi and MQTT connection
    led.value(0)
    connectWIFI()
    c = MQTTClient(MQTT_CLIENT_ID, MQTT_SERVER, keepalive=60)
    connectMQTT(c)

    while True:

        if not network.WLAN(network.STA_IF).isconnected():
            print("WiFi disconnected, reconnecting...")
            connectWIFI()

        if not mqtt_connected:
            print("MQTT disconnected, reconnecting...")
            connectMQTT(c)

        if (time.time() - last_message) > message_interval:
            temp, hum = read_sensor()
            print(temp)
            print(hum)
            if mqtt_connected:
                c.publish(MQTT_TOPIC_TEMPHUM, str(temp)+","+str(hum), retain=False, qos=1)
                last_message = time.time()
            else:
                print("MQTT connection lost, unable to publish message")
            
finally:
    print("Error! Disconnecting and resetting...")
    try:
        c.disconnect()
    except OSError as e:
        print("Error disconnecting from MQTT broker:", e)
    reset()  # Restart ESP8266 in case of an error
